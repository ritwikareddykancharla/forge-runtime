@id("permit-prod-mutation-for-ops-agents")
permit(
  principal,
  action in [Action::"ApplyProdPatch", Action::"MutateProdConfig", Action::"DropTable"],
  resource in Environment::"prod"
) when { principal.role == "ops-remediator" || principal.role == "admin" };

@id("forbid-destructive-ddl-unconditionally")

  principal,

  resource
);

  principal,
  action in [Action::"ApplyProdPatch", Action::"MutateProdConfig"],
  resource in Environment::"prod"
